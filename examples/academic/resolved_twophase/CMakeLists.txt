cmake_minimum_required(VERSION 3.22)
project(ResolvedTwoPhase LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(MYPELE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(AMReX_SPACEDIM 2 CACHE STRING "Cartesian dimension")
set(AMReX_MPI ON CACHE BOOL "MPI parallelism")
set(AMReX_OMP OFF CACHE BOOL "OpenMP parallelism")
set(AMReX_PARTICLES OFF CACHE BOOL "Resolved interfaces have no subgrid particles")
set(AMReX_EB OFF CACHE BOOL "No embedded geometry in canonical interface benchmarks")
set(AMReX_FORTRAN OFF CACHE BOOL "No Fortran required")
set(AMReX_LINEAR_SOLVERS OFF CACHE BOOL "Explicit compressible solver")
set(AMReX_AMRLEVEL OFF CACHE BOOL "Driver uses MultiFab")
set(AMReX_GPU_BACKEND NONE CACHE STRING "NONE or CUDA")
if(AMReX_GPU_BACKEND STREQUAL "CUDA")
    enable_language(CUDA)
endif()
add_subdirectory("${MYPELE_ROOT}/external/amrex" amrex)
set(PHYSICS "${MYPELE_ROOT}/external/PeleC/Submodules/PelePhysics")
add_executable(resolved_twophase main.cpp)
target_link_libraries(resolved_twophase PRIVATE AMReX::amrex)
target_include_directories(resolved_twophase PRIVATE
    "${MYPELE_ROOT}/src/academic/twophase" "${MYPELE_ROOT}/src/academic/vof"
    "${PHYSICS}/Source" "${PHYSICS}/Source/Eos" "${PHYSICS}/Source/Utility" "${PHYSICS}/Mechanisms/LiDryer")
if(AMReX_GPU_BACKEND STREQUAL "CUDA")
    setup_target_for_cuda_compilation(resolved_twophase)
endif()

include(CTest)
if(BUILD_TESTING AND AMReX_GPU_BACKEND STREQUAL "NONE")
    add_executable(test_mechanical_eos_plic "${MYPELE_ROOT}/test/unit/test_mechanical_eos_plic.cpp")
    target_link_libraries(test_mechanical_eos_plic PRIVATE AMReX::amrex)
    target_include_directories(test_mechanical_eos_plic PRIVATE
        "${MYPELE_ROOT}/src/academic/twophase" "${MYPELE_ROOT}/src/academic/vof"
        "${PHYSICS}/Source" "${PHYSICS}/Source/Eos" "${PHYSICS}/Source/Utility"
        "${PHYSICS}/Mechanisms/LiDryer")
    add_test(NAME mechanical_eos_and_plic COMMAND test_mechanical_eos_plic)
endif()
